cmake_minimum_required(VERSION 3.20)
project(MouseSide VERSION 1.0.0 LANGUAGES CXX)
enable_language(RC)
if(NOT WIN32)
  message(FATAL_ERROR "MouseSide targets Windows; use a Windows toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive- /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion $<$<CONFIG:Release>:-Os> -ffunction-sections -fdata-sections)
  add_link_options(-static -static-libgcc -static-libstdc++)
endif()
add_library(mouseside_core src/Model.cpp src/Config.cpp)
target_include_directories(mouseside_core PUBLIC src)
target_link_libraries(mouseside_core PUBLIC user32 shell32 advapi32 ole32 uuid)
include(CTest)
add_library(mouseside_input src/Input.cpp)
target_link_libraries(mouseside_input PUBLIC mouseside_core)
add_executable(MouseSide WIN32 src/main.cpp src/App.cpp res/app.rc)
target_include_directories(MouseSide PRIVATE res)
target_link_libraries(MouseSide PRIVATE mouseside_input comctl32 gdi32 gdiplus dwmapi uxtheme)
if(MSVC)
  target_link_options(MouseSide PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_link_options(MouseSide PRIVATE -municode -Wl,--gc-sections -s)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE mouseside_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(input_tests tests/input_tests.cpp)
  target_link_libraries(input_tests PRIVATE mouseside_input)
  add_test(NAME input COMMAND input_tests)
  add_executable(gui_tests tests/gui_tests.cpp)
  target_link_libraries(gui_tests PRIVATE mouseside_core)
  if(MINGW)
    target_link_options(gui_tests PRIVATE -municode)
  endif()
  add_test(NAME gui COMMAND gui_tests $<TARGET_FILE:MouseSide>)
endif()
